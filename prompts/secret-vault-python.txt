"""
SoftPlay Auto Password Encryption — decrypts credentials that were
encrypted on the developer's machine before this file was generated (see
the Mandatory Standard above). Algorithm: AES-256-CTR + HMAC-SHA256
(Encrypt-then-MAC). AES-256 itself is implemented here in pure Python (no
pip install required — this project never assumes PyPI network access);
HMAC-SHA256 and base64 come from the standard library. The master key
never appears in this file: it is supplied at test-run time via the
SoftPlay_SECRET_KEY environment variable (base64) — SoftPlay's own
"Verify & Fix Code" sets this automatically; a standalone CI/CD pipeline
running this file later must supply it the same way, sourced from your
organization's own secrets manager (Vault, CyberArk, Azure Key Vault, AWS
Secrets Manager, ...).
"""
import base64
import hashlib
import hmac
import os

_ENV_VAR = "SoftPlay_SECRET_KEY"


def _gf_mul(a, b):
    p = 0
    for _ in range(8):
        if b & 1:
            p ^= a
        hi = a & 0x80
        a = (a << 1) & 0xFF
        if hi:
            a ^= 0x1B
        b >>= 1
    return p


def _gf_inv(a):
    if a == 0:
        return 0
    for x in range(1, 256):
        if _gf_mul(a, x) == 1:
            return x
    return 0


def _build_sbox():
    sbox = [0] * 256
    for a in range(256):
        r = b = _gf_inv(a)
        for _ in range(4):
            r = ((r << 1) | (r >> 7)) & 0xFF
            b ^= r
        sbox[a] = b ^ 0x63
    return sbox


_SBOX = _build_sbox()
_RCON = [0x01]
for _i in range(1, 14):
    _RCON.append(_gf_mul(_RCON[-1], 2))


def _key_expansion_256(key):
    nk, nr = 8, 14
    w = [list(key[4 * i:4 * i + 4]) for i in range(nk)]
    for i in range(nk, 4 * (nr + 1)):
        temp = list(w[i - 1])
        if i % nk == 0:
            temp = temp[1:] + temp[:1]
            temp = [_SBOX[b] for b in temp]
            temp[0] ^= _RCON[i // nk - 1]
        elif nk > 6 and i % nk == 4:
            temp = [_SBOX[b] for b in temp]
        w.append([w[i - nk][j] ^ temp[j] for j in range(4)])
    return [b"".join(bytes(w[r * 4 + c]) for c in range(4)) for r in range(nr + 1)]


def _add_round_key(state, rk):
    return bytes(s ^ k for s, k in zip(state, rk))


def _sub_bytes(state):
    return bytes(_SBOX[b] for b in state)


def _shift_rows(state):
    out = [0] * 16
    for c in range(4):
        for r in range(4):
            out[r + 4 * c] = state[r + 4 * ((c + r) % 4)]
    return bytes(out)


def _mix_columns(state):
    out = [0] * 16
    for c in range(4):
        a0, a1, a2, a3 = state[4 * c:4 * c + 4]
        out[4 * c + 0] = _gf_mul(a0, 2) ^ _gf_mul(a1, 3) ^ a2 ^ a3
        out[4 * c + 1] = a0 ^ _gf_mul(a1, 2) ^ _gf_mul(a2, 3) ^ a3
        out[4 * c + 2] = a0 ^ a1 ^ _gf_mul(a2, 2) ^ _gf_mul(a3, 3)
        out[4 * c + 3] = _gf_mul(a0, 3) ^ a1 ^ a2 ^ _gf_mul(a3, 2)
    return bytes(out)


def _aes256_encrypt_block(key, block):
    round_keys = _key_expansion_256(key)
    state = _add_round_key(block, round_keys[0])
    for rnd in range(1, 14):
        state = _sub_bytes(state)
        state = _shift_rows(state)
        state = _mix_columns(state)
        state = _add_round_key(state, round_keys[rnd])
    state = _sub_bytes(state)
    state = _shift_rows(state)
    return _add_round_key(state, round_keys[14])


def _aes256_ctr_crypt(key, iv, data):
    out = bytearray()
    counter = int.from_bytes(iv, "big")
    i = 0
    while len(out) < len(data):
        keystream = _aes256_encrypt_block(key, counter.to_bytes(16, "big"))
        chunk = data[i:i + 16]
        out += bytes(a ^ b for a, b in zip(chunk, keystream))
        counter = (counter + 1) % (1 << 128)
        i += 16
    return bytes(out)


def _hmac_sha256(key, *parts):
    h = hmac.new(key, digestmod=hashlib.sha256)
    for p in parts:
        h.update(p)
    return h.digest()


def decrypt_secret(token: str) -> str:
    """Decrypts a SoftPlay ENC[v1:...] token — see the module docstring above."""
    if not token or not token.startswith("ENC[") or not token.endswith("]"):
        raise ValueError("Not a SoftPlay encrypted token: %r" % (token,))
    parts = token[4:-1].split(":")
    if len(parts) != 4 or parts[0] != "v1":
        raise ValueError("Unsupported SoftPlay token version: %r" % (token,))
    iv = base64.b64decode(parts[1])
    tag = base64.b64decode(parts[2])
    ciphertext = base64.b64decode(parts[3])

    master_key_b64 = os.environ.get(_ENV_VAR)
    if not master_key_b64:
        raise EnvironmentError(
            "Missing %s environment variable — SoftPlay's Verify & Fix Code sets this "
            "automatically; a standalone run must supply it from your organization's secrets manager." % _ENV_VAR
        )
    master_key = base64.b64decode(master_key_b64)

    enc_key = _hmac_sha256(master_key, b"SoftPlay-aes-key-v1")
    mac_key = _hmac_sha256(master_key, b"SoftPlay-hmac-key-v1")

    expected_tag = _hmac_sha256(mac_key, iv, ciphertext)
    if not hmac.compare_digest(expected_tag, tag):
        raise ValueError("SoftPlay encrypted credential failed its integrity check — token may be corrupted or tampered with.")

    plaintext = _aes256_ctr_crypt(enc_key, iv, ciphertext)
    return plaintext.decode("utf-8")
